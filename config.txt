# config.txt - Configuracion del firmware de la Raspberry Pi 3B
#
# Lo lee la GPU antes de arrancar la CPU ARM. Va en la particion FAT32 de la
# SD, junto al firmware y a kernel8.img.

# Arrancar la CPU en AArch64 (y buscar kernel8.img en vez de kernel.img)
arm_64bit=1

# Mantener la UART activa y con reloj estable
enable_uart=1

# Liberar la PL011 (UART0) del modulo Bluetooth y dejarla en los pines
# GPIO 14/15 del conector. Sin esto, la PL011 esta cableada al Bluetooth y
# por el conector sale la mini-UART, que es otro periferico distinto.
dtoverlay=disable-bt

# Reloj de la PL011. El divisor de baudios de src/uart.c (IBRD=26, FBRD=3)
# esta calculado para 48 MHz; dejarlo explicito evita depender del valor
# por defecto del firmware.
init_uart_clock=48000000

# Darle a la GPU el minimo: mas RAM para nosotros. El kernel pregunta por el
# buzon cuanta hay de verdad, asi que cualquier valor funciona.
gpu_mem=16

# Arrancar cuanto antes
boot_delay=0
disable_splash=1

kernel=kernel8.img
